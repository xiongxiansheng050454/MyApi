CREATE TABLE user_daily_stats (
    user_id BIGINT NOT NULL,
    stat_date DATE NOT NULL,
    
    -- 用量汇总
    total_input_tokens BIGINT DEFAULT 0,
    total_output_tokens BIGINT DEFAULT 0,
    total_cached_input_tokens BIGINT DEFAULT 0,
    total_tokens BIGINT DEFAULT 0, -- 可以直接SUM生成列，但存下来查询更快
    total_cost DECIMAL(12, 8) DEFAULT 0.00000000,
    
    -- 请求质量
    request_count INT DEFAULT 0,
    success_count INT DEFAULT 0,
    error_count INT DEFAULT 0,
    
    -- 元数据（记录最后更新的明细ID，便于排查）
    last_processed_id BIGINT DEFAULT 0,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    
    PRIMARY KEY (user_id, stat_date)
);

CREATE TABLE model_pricing (
    id BIGSERIAL PRIMARY KEY,
    
    -- 关联维度
    channel_id BIGINT NOT NULL REFERENCES channels(id) ON DELETE CASCADE,
    model_name VARCHAR(100) NOT NULL,
    
    -- 核心价格（单位：美元 / 1M tokens）
    input_price_per_1m DECIMAL(12, 8) NOT NULL DEFAULT 0,
    output_price_per_1m DECIMAL(12, 8) NOT NULL DEFAULT 0,
    
    -- Prompt Caching 折扣价（若无则与 input_price_per_1m 一致）
    cached_input_price_per_1m DECIMAL(12, 8) DEFAULT NULL,
    
    -- 货币类型
    currency VARCHAR(3) NOT NULL DEFAULT 'USD',
    
    -- 审计
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_pricing_channel_model 
        FOREIGN KEY (channel_id, model_name) 
        REFERENCES channel_models(channel_id, model_name) 
        ON DELETE CASCADE,

    -- 约束
    CONSTRAINT unique_pricing_period UNIQUE (channel_id, model_name),
    CONSTRAINT check_pricing_positive CHECK (input_price_per_1m >= 0 AND output_price_per_1m >= 0)
);

CREATE INDEX idx_pricing ON model_pricing (channel_id, model_name);

CREATE TABLE rate_limit_rules (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    rule_name VARCHAR(100) NOT NULL COMMENT '规则名称，便于管理',
    
    -- 目标对象：global, user, api_key, model,channel
    target_type VARCHAR(20) NOT NULL,
    -- 目标值：* 代表通配；具体值存储 user_id、api_key 等
    target_value TEXT NOT NULL DEFAULT '*',
    
    -- 限流指标：rpm(每分钟请求), tpm(每分钟Token), rpd(每日请求), tpd(每日Token), concurrency(并发)
    metric VARCHAR(20) NOT NULL,
    limit_value INTEGER NOT NULL COMMENT '限额数值',
    window_seconds INTEGER NOT NULL COMMENT '时间窗口(秒)，如 60, 86400',
    
    -- 动作与优先级
    action VARCHAR(20) DEFAULT 'reject' COMMENT 'reject 或 queue',
    priority INTEGER DEFAULT 0 COMMENT '数值越小优先级越高',
    
    enabled BOOLEAN DEFAULT TRUE,
    description TEXT,
    
    -- JSONB 扩展字段，用于存放特殊配置（如排队超时时间、自定义错误码）
    extras JSONB DEFAULT '{}'::jsonb,
    
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    
    -- 约束：防止乱填 target_type
    CONSTRAINT check_target_type CHECK (target_type IN ('global', 'user', 'api_key', 'model', 'channel')),
    -- 约束：防止乱填 metric
    CONSTRAINT check_metric CHECK (metric IN ('rpm', 'tpm', 'rpd', 'tpd', 'concurrency'))
);

CREATE INDEX idx_rate_limit_lookup ON rate_limit_rules (target_type, target_value, metric, priority) WHERE enabled = true;

CREATE TABLE usage_logs (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    
    request_id VARCHAR(64) NOT NULL UNIQUE COMMENT '网关唯一请求ID',
    user_id BIGINT NOT NULL,
    api_key_id BIGINT NOT NULL,
    channel_id BIGINT NOT NULL,
    
    model VARCHAR(100) NOT NULL,
    upstream_model VARCHAR(100) DEFAULT NULL,
    
    input_tokens INT NOT NULL DEFAULT 0,
    output_tokens INT NOT NULL DEFAULT 0,
    cached_input_tokens INT DEFAULT 0,
    total_tokens INT GENERATED ALWAYS AS (input_tokens + output_tokens) STORED,
    
    -- 费用快照（单价单位：美元/1M tokens）
    unit_price_input_per_1m DECIMAL(12, 8) NOT NULL COMMENT '输入单价（$/1M tokens）',
    unit_price_output_per_1m DECIMAL(12, 8) NOT NULL COMMENT '输出单价（$/1M tokens）',
    total_cost DECIMAL(12, 8) NOT NULL COMMENT '本次请求总费用（美元）',
    
    duration_ms INT NOT NULL,
    ttft_ms INT DEFAULT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'success',
    error_code VARCHAR(50) DEFAULT NULL,
    
    client_ip INET DEFAULT NULL,
    extra JSONB DEFAULT '{}'::jsonb,
    
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

- 最常用的用户查询（查某个用户某段时间的账单）
CREATE INDEX CONCURRENTLY idx_usage_user_created ON usage_logs (user_id, created_at DESC);

-- 统计渠道消耗（查某个上游花了多少钱）
CREATE INDEX CONCURRENTLY idx_usage_channel_created ON usage_logs (channel_id, created_at DESC);

-- 按API Key审计（查哪个Key消耗最多）
CREATE INDEX CONCURRENTLY idx_usage_key_created ON usage_logs (api_key_id, created_at DESC);

-- 状态筛选（查最近失败的请求）
CREATE INDEX CONCURRENTLY idx_usage_status_created ON usage_logs (status, created_at DESC);

CREATE INDEX idx_usage_user_model_cost ON usage_logs (user_id, created_at DESC) INCLUDE (model, input_tokens, output_tokens, total_cost);

CREATE TABLE users (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    
    password_hash TEXT NOT NULL COMMENT 'bcrypt/Argon2 哈希，不可逆',
    
    -- 业务属性
    user_group VARCHAR(20) NOT NULL DEFAULT 'default' COMMENT '分组：default/vip/enterprise，用于路由和限流',
    status VARCHAR(20) NOT NULL DEFAULT 'active' COMMENT 'active/suspended/deleted',
    
    -- 辅助信息
    nickname VARCHAR(100) DEFAULT NULL,
    last_login_at TIMESTAMPTZ DEFAULT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 按状态过滤时有用
CREATE INDEX idx_users_status ON users (status);

CREATE TABLE balance_transactions (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id BIGINT NOT NULL,
    
    -- 变动金额（正=充值/退款，负=消费/扣减）
    amount DECIMAL(12, 6) NOT NULL, 
    
    -- 变动前后的快照（方便对账，无需反算）
    balance_before DECIMAL(12, 6) NOT NULL,
    balance_after DECIMAL(12, 6) NOT NULL,
    
    -- 业务类型：recharge, consume, refund, freeze, unfreeze
    tx_type VARCHAR(20) NOT NULL,
    
    -- 关联外部业务ID
    related_request_id VARCHAR(64) NULL COMMENT '关联的请求ID（溯源哪次调用扣的钱）',
    related_order_id VARCHAR(64) NULL COMMENT '关联的充值订单号',
    
    description TEXT,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP, -- 不可变，不允许UPDATE
    
    CONSTRAINT check_amount_non_zero CHECK (amount != 0)
);

-- 按用户和时间查询流水
CREATE INDEX idx_tx_user_created ON balance_transactions (user_id, created_at DESC);

CREATE TABLE user_balances (
    user_id BIGINT PRIMARY KEY,          -- 与 client_api_keys 中的 user_id 对应
    available_balance DECIMAL(12, 6) NOT NULL DEFAULT 0 COMMENT '可用余额（美元/积分）',
    frozen_balance DECIMAL(12, 6) NOT NULL DEFAULT 0 COMMENT '冻结余额（处理中的请求）',
    
    -- 乐观锁版本号，更新时检查 version = old_version
    version BIGINT NOT NULL DEFAULT 1,
    
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT check_balance_non_negative CHECK (available_balance >= 0)
);

CREATE TABLE client_api_keys (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id BIGINT NOT NULL COMMENT '所属用户/租户ID',
    key_name VARCHAR(100) NOT NULL DEFAULT 'default' COMMENT '密钥备注名（便于管理）',
    
    -- 核心安全字段
    key_prefix VARCHAR(10) NOT NULL COMMENT '前缀（如 sk-），用于快速识别来源',
    key_hash TEXT NOT NULL COMMENT '存储哈希值（如 bcrypt/Argon2），绝不存明文',
    
    -- 权限与作用域（JSONB 灵活控制）
    permissions JSONB DEFAULT '{"models": ["*"]}'::jsonb COMMENT '允许调用的模型列表，["*"]代表全部',
    
    -- 限流覆盖（覆盖通用限流规则）
    rate_limit_overrides JSONB DEFAULT '{}'::jsonb COMMENT '例如：{"rpm": 500, "tpm": 100000}',
    
    -- 状态与时间
    expires_at TIMESTAMPTZ DEFAULT NULL COMMENT '过期时间（NULL代表永不过期）',
    is_active BOOLEAN DEFAULT TRUE,
    last_used_at TIMESTAMPTZ DEFAULT NULL COMMENT '最后调用时间',
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    -- 约束与索引
    CONSTRAINT unique_user_keyname UNIQUE (user_id, key_name),
);

-- 关键索引：只索引激活且未过期的key，大幅提升验证速度
CREATE INDEX idx_api_keys_auth ON client_api_keys (key_prefix, key_hash) WHERE is_active = true AND (expires_at IS NULL OR expires_at > CURRENT_TIMESTAMP);

--  部分索引，只索引启用的规则，查询更快且节省空间
CREATE INDEX idx_enabled_priority ON rate_limit_rules (priority ASC) WHERE enabled = true;


CREATE TABLE channels (
    id              BIGSERIAL PRIMARY KEY,
    name            VARCHAR(100) NOT NULL,
    base_url        VARCHAR(255) NOT NULL,
    api_key         VARCHAR(500) NOT NULL,          -- 存储加密后的密文
    auth_type       VARCHAR(20) DEFAULT 'bearer',
    extra_config    JSONB,
    status          SMALLINT DEFAULT 1 NOT NULL,    -- 1:启用 0:停用
    weight          INT DEFAULT 100,                -- 负载均衡权重（同优先级下使用）
    priority        INT DEFAULT 0,                  -- 路由优先级（数值越大越优先，默认0）
    created_at      TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at      TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_channels_active_priority ON channels (priority DESC) WHERE status = 1;

CREATE TABLE channel_models (
    id                BIGSERIAL PRIMARY KEY,
    channel_id        BIGINT NOT NULL REFERENCES channels(id) ON DELETE CASCADE,
    model_name        VARCHAR(100) NOT NULL,          -- 网关对外暴露的模型名（如 gpt-4）
    upstream_model    VARCHAR(100) NOT NULL,          -- 上游实际模型名（如 gpt-4-0613）
    enabled           BOOLEAN DEFAULT TRUE,
    created_at        TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at        TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(channel_id, model_name)                   -- 同一渠道下不重复映射同一网关模型
);

CREATE INDEX idx_channel_models_model_name ON channel_models(model_name);

CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- rate_limit_rules 表
DROP TRIGGER IF EXISTS trigger_update_rate_limit_updated_at ON rate_limit_rules;
CREATE TRIGGER trigger_update_rate_limit_updated_at
    BEFORE UPDATE ON rate_limit_rules
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();
